CREATE USER C##GROO IDENTIFIED BY java;
GRANT CONNECT, RESOURCE TO C##GROO
ALTER USER C##GROO QUOTA UNLIMITED ON USERS;


//Groo 에서 테이블 생성
CREATE TABLE users (
  id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name VARCHAR2(50),
  user_id VARCHAR2(50),
  password VARCHAR2(100)
);

CREATE TABLE team (
  TEAM_NAME VARCHAR2(50),
  user_id VARCHAR2(50),
  TEAM_INFO VARCHAR2(100)
);

CREATE TABLE team_members (
  TEAM_NAME VARCHAR2(50),
  user_id VARCHAR2(50),
  TEAM_ROLE VARCHAR2(10)
);
